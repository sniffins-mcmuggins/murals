# Admin Panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build an internal API for admins to manage users, issue access grants, and create/distribute promo codes that bypass billing.

**Architecture:** New `api/internal/admin` package behind a `RequireAdmin` middleware (role=admin + mfa_enabled=true). A new `access_grants` table and `promo_codes` table store free-access records; three existing billing checks gain a fallback to these tables. User-facing promo redemption lives at `POST /promo/redeem`.

**Tech Stack:** Go, chi router, pgx/sqlc, testcontainers (existing patterns)

```sql
-- db/migrations/000005_admin.up.sql

-- Promo codes: distributable codes granting free platform access.
-- max_uses NULL = unlimited. expires_at NULL = never expires.
CREATE TABLE promo_codes (
    id            uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
    code          text        UNIQUE NOT NULL,
    plan          text        NOT NULL,
    duration_days integer     NOT NULL,
    max_uses      integer,
    use_count     integer     NOT NULL DEFAULT 0,
    expires_at    timestamptz,
    created_by    uuid        REFERENCES users(id) ON DELETE SET NULL,
    revoked_at    timestamptz,
    created_at    timestamptz NOT NULL DEFAULT now()
);

-- Access grants: free-access records for a user+plan, sourced either from a
-- direct admin grant (granted_by set) or a promo code redemption (promo_code_id set).
-- festival_id is required when plan = 'festival_activation'.
CREATE TABLE access_grants (
    id            uuid        PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id       uuid        NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    plan          text        NOT NULL,
    festival_id   uuid        REFERENCES festivals(id) ON DELETE SET NULL,
    valid_until   timestamptz NOT NULL,
    granted_by    uuid        REFERENCES users(id) ON DELETE SET NULL,
    promo_code_id uuid        REFERENCES promo_codes(id) ON DELETE SET NULL,
    note          text,
    revoked_at    timestamptz,
    created_at    timestamptz NOT NULL DEFAULT now()
);

CREATE INDEX access_grants_user_idx  ON access_grants (user_id);
CREATE INDEX access_grants_plan_idx  ON access_grants (user_id, plan);
```

```sql
-- db/queries/admin.sql

-- name: ListUsers :many
SELECT id, email, role, mfa_enabled, created_at
FROM users
WHERE (sqlc.arg(email)::text = '' OR email ILIKE '%' || sqlc.arg(email)::text || '%')
ORDER BY created_at DESC
LIMIT sqlc.arg(lim)::int
OFFSET sqlc.arg(off)::int;

-- name: CreateAccessGrant :one
INSERT INTO access_grants (user_id, plan, festival_id, valid_until, granted_by, promo_code_id, note)
VALUES ($1, $2, $3, $4, $5, $6, $7)
RETURNING *;

-- name: RevokeAccessGrant :exec
UPDATE access_grants
SET revoked_at = now()
WHERE id = $1 AND revoked_at IS NULL;

-- name: HasActiveGrant :one
SELECT EXISTS (
    SELECT 1 FROM access_grants
    WHERE user_id = $1
      AND plan = $2
      AND revoked_at IS NULL
      AND valid_until > now()
      AND (festival_id IS NULL OR festival_id = $3)
) AS has_grant;

-- name: ListActiveGrants :many
SELECT * FROM access_grants
WHERE user_id = $1 AND revoked_at IS NULL AND valid_until > now()
ORDER BY created_at DESC;

-- name: CreatePromoCode :one
INSERT INTO promo_codes (code, plan, duration_days, max_uses, expires_at, created_by)
VALUES ($1, $2, $3, $4, $5, $6)
RETURNING *;

-- name: GetPromoCodeByCode :one
SELECT * FROM promo_codes WHERE code = $1;

-- name: GetPromoCodeByID :one
SELECT * FROM promo_codes WHERE id = $1;

-- name: ListPromoCodes :many
SELECT * FROM promo_codes ORDER BY created_at DESC;

-- name: RevokePromoCode :exec
UPDATE promo_codes
SET revoked_at = now()
WHERE id = $1 AND revoked_at IS NULL;

-- name: IncrementPromoUseCount :exec
UPDATE promo_codes SET use_count = use_count + 1 WHERE id = $1;

-- name: HasRedeemedPromo :one
SELECT EXISTS (
    SELECT 1 FROM access_grants
    WHERE user_id = $1 AND promo_code_id = $2
) AS has_redeemed;
```

```go
package admin

import (
	"time"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5/pgtype"
)

func pgUUIDFromString(s string) (pgtype.UUID, error) {
	parsed, err := uuid.Parse(s)
	if err != nil {
		return pgtype.UUID{}, err
	}
	return pgtype.UUID{Bytes: [16]byte(parsed), Valid: true}, nil
}

func pgUUIDNullable(s string) pgtype.UUID {
	if s == "" {
		return pgtype.UUID{}
	}
	v, err := pgUUIDFromString(s)
	if err != nil {
		return pgtype.UUID{}
	}
	return v
```

```go
	"log/slog"

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"

	"github.com/sniffins-mcmuggins/render/api/internal/auth"
	"github.com/sniffins-mcmuggins/render/api/internal/httperr"
	"github.com/sniffins-mcmuggins/render/api/internal/sqlcdb"
)

// RequireAdmin gates routes to admin users who have completed MFA enrollment.
// Must sit downstream of auth.Middleware in the handler chain.
//
// Checks:
//  1. Valid Principal exists → 401 if anonymous
//  2. role == "admin" → 403 if wrong role
//  3. user.mfa_enabled == true in DB → 403 if TOTP not enrolled
//
// Check 3 is a safety net: if mfa_enabled=true the login flow forced the
// mfa_pending → /auth/mfa/verify path, so a full-scope token implies TOTP was verified.
func RequireAdmin(pool *pgxpool.Pool) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			principal, err := auth.User(r.Context())
			if err != nil {
				httperr.Unauthorized(w)
				return
			}
			if principal.Role != "admin" {
				httperr.Forbidden(w)
				return
			}
			userUUID, err := pgUUIDFromString(principal.UserID)
			if err != nil {
				httperr.InternalServerError(w)
				return
			}
			q := sqlcdb.New(pool)
			user, err := q.GetUserByID(r.Context(), userUUID)
			if err != nil {
				if err == pgx.ErrNoRows {
					httperr.Unauthorized(w)
					return
				}
				slog.Error("require admin: user lookup failed", "err", err, "user_id", principal.UserID)
				httperr.InternalServerError(w)
				return
			}
			if !user.MfaEnabled {
				httperr.Write(w, http.StatusForbidden, "Forbidden", "admin account must have MFA enrolled")
				return
			}
			next.ServeHTTP(w, r)
		})
	}
}
```

```go
			httperr.BadRequest(w, "invalid request body")

		}
		if !validPlans[req.Plan] {

			return
		}

			httperr.BadRequest(w, "festival_id is required for festival_activation grants")

		}

		var note *string
		if req.Note != "" {

		}

		if _, err := q.GetUserByID(r.Context(), userUUID); err != nil {
			if errors.Is(err, pgx.ErrNoRows) {

				return
			}

			httperr.InternalServerError(w)
			return
		}

		grant, err := q.CreateAccessGrant(r.Context(), sqlcdb.CreateAccessGrantParams{
			UserID:      userUUID,
			Plan:        req.Plan,
			FestivalID:  pgUUIDNullable(req.FestivalID),
			ValidUntil:  pgTimestamptz(time.Now().Add(time.Duration(req.DurationDays) * 24 * time.Hour)),
			GrantedBy:   adminUUID,
			PromoCodeID: pgUUIDNullable(""),
			Note:        note,
		})

			slog.Error("admin: create grant", "err", err)
			httperr.InternalServerError(w)
			return
		}

		_ = json.NewEncoder(w).Encode(resp)
```
```go
	}

// RedeemPromoHandler handles POST /promo/redeem (user-facing, normal auth only).
// Validates the code and creates an access grant for the authenticated user.
func RedeemPromoHandler(pool *pgxpool.Pool) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		principal, err := auth.User(r.Context())
		if err != nil {
			httperr.Unauthorized(w)
			return
		}
		userUUID, err := pgUUIDFromString(principal.UserID)
		if err != nil {
			httperr.InternalServerError(w)
			return
		}

		var req redeemRequest
		if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
			httperr.BadRequest(w, "invalid request body")
			return
		}
		if req.Code == "" {
			httperr.BadRequest(w, "code is required")
			return
		}

		q := sqlcdb.New(pool)
		pc, err := q.GetPromoCodeByCode(r.Context(), req.Code)
		if err != nil {
			if errors.Is(err, pgx.ErrNoRows) {
				httperr.NotFound(w)
				return
			}
			slog.Error("redeem promo: lookup code", "err", err)
			httperr.InternalServerError(w)
			return
		}

		if pc.RevokedAt.Valid {
			httperr.Write(w, http.StatusGone, "Gone", "this promo code has been revoked")
			return
		}
		if pc.ExpiresAt.Valid && time.Now().After(pc.ExpiresAt.Time) {
			httperr.Write(w, http.StatusGone, "Gone", "this promo code has expired")
			return
		}
		if pc.MaxUses != nil && pc.UseCount >= *pc.MaxUses {
			httperr.Write(w, http.StatusConflict, "Conflict", "this promo code has reached its usage limit")
			return
		}

		alreadyRedeemed, err := q.HasRedeemedPromo(r.Context(), sqlcdb.HasRedeemedPromoParams{
			UserID:      userUUID,
			PromoCodeID: pc.ID,
		})
		if err != nil {
			slog.Error("redeem promo: check existing redemption", "err", err)
			httperr.InternalServerError(w)
			return
		}
		if alreadyRedeemed {
			httperr.Write(w, http.StatusConflict, "Conflict", "you have already redeemed this promo code")
			return
		}

		grant, err := q.CreateAccessGrant(r.Context(), sqlcdb.CreateAccessGrantParams{
			UserID:      userUUID,
			Plan:        pc.Plan,
			ValidUntil:  pgTimestamptz(time.Now().Add(time.Duration(pc.DurationDays) * 24 * time.Hour)),
			PromoCodeID: pc.ID,
		})
		if err != nil {
			slog.Error("redeem promo: create grant", "err", err)
			httperr.InternalServerError(w)
			return
		}

		if err := q.IncrementPromoUseCount(r.Context(), pc.ID); err != nil {
			slog.Error("redeem promo: increment use count", "err", err, "code_id", pc.ID.String())
			// Non-fatal: grant was created, counter off by one is acceptable.
		}

		resp := grantResponse{
			ID:         grant.ID.String(),
			UserID:     grant.UserID.String(),
			Plan:       grant.Plan,
			ValidUntil: grant.ValidUntil.Time.Format(time.RFC3339),
```

```go
package admin

import (
	"errors"
	"time"

	"github.com/google/uuid"
	"github.com/jackc/pgerrcode"
	"github.com/jackc/pgx/v5/pgconn"
	"github.com/jackc/pgx/v5/pgtype"
)

func pgUUIDFromString(s string) (pgtype.UUID, error) {
	parsed, err := uuid.Parse(s)
	if err != nil {
		return pgtype.UUID{}, err
	}
	return pgtype.UUID{Bytes: [16]byte(parsed), Valid: true}, nil
}

func pgUUIDNullable(s string) pgtype.UUID {
	if s == "" {
		return pgtype.UUID{}
	}
	v, err := pgUUIDFromString(s)
	if err != nil {
		return pgtype.UUID{}
	}
	return v
}
```
